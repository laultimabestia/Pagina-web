// ================================================== // CONFIGURACIÓN - AJUSTA LOS NOMBRES DE TUS HOJAS // ================================================== const HOJA_DATOS = "DATOS"; const HOJA_PREMIOS = "PREMIOS"; const HOJA_CANJES = "CANJES"; // ================================================== // PÁGINA WEB RESPONSIVE (GET) - DISEÑO MODERNO CON LOGO // ================================================== function doGet() { return HtmlService.createHtmlOutput(` Consulta de Puntos

🎓 Consulta de Puntos

Ingresa tu RUN (solo números)

Datos actualizados al instante
`).setTitle("Consulta de Puntos"); } // ================================================== // FUNCIÓN SERVIDORA PARA LA WEB (usa la lógica completa) // ================================================== function consultarPadre(data) { const resultado = consultarAlumno(data); const respuestaJson = JSON.parse(resultado.getContent()); return respuestaJson; } // ================================================== // ENDPOINT PARA PYTHON (POST) // ================================================== function doPost(e) { try { const data = JSON.parse(e.postData.contents); const comando = data.comando; if (comando === "sumar") return sumarPuntos(data); if (comando === "consultar") return consultarAlumno(data); if (comando === "canjear") return canjearPremio(data); if (comando === "listar_canjes") return listarCanjes(); // COMANDOS PARA GESTIÓN DE ALUMNOS if (comando === "listar_alumnos") return listarAlumnos(); if (comando === "agregar_alumno") return agregarAlumno(data); if (comando === "editar_alumno") return editarAlumno(data); if (comando === "eliminar_alumno") return eliminarAlumno(data); // NUEVOS COMANDOS PARA GESTIÓN DE PREMIOS if (comando === "listar_premios") return listarPremios(); if (comando === "agregar_premio") return agregarPremio(data); if (comando === "editar_premio") return editarPremio(data); if (comando === "eliminar_premio") return eliminarPremio(data); return respuesta(false, "Comando desconocido"); } catch (error) { return respuesta(false, error.toString()); } } // ================================================== // FUNCIONES PARA PYTHON // ================================================== function sumarPuntos(data) { const rut = data.rut.toString().replace(/\D/g, ""); const puntosSumar = Number(data.puntos) || 50; const ss = SpreadsheetApp.getActive(); const hoja = ss.getSheetByName(HOJA_DATOS); const filas = hoja.getDataRange().getValues(); for (let i = 1; i < filas.length; i++) { let rutTabla = filas[i][1].toString().replace(/\D/g, ""); if (rutTabla === rut) { let puntosActual = Number(filas[i][5]) || 0; let nuevosPuntos = puntosActual + puntosSumar; hoja.getRange(i+1, 6).setValue(nuevosPuntos); let nombre = `${filas[i][2]} ${filas[i][3]} ${filas[i][4]}`.trim(); return respuesta(true, "Puntos sumados", { nombre: nombre, puntos: nuevosPuntos }); } } return respuesta(false, "RUT no encontrado"); } function consultarAlumno(data) { const rut = data.rut.toString().replace(/\D/g, ""); const ss = SpreadsheetApp.getActive(); const hojaDatos = ss.getSheetByName(HOJA_DATOS); const filas = hojaDatos.getDataRange().getValues(); for (let i = 1; i < filas.length; i++) { let rutTabla = filas[i][1].toString().replace(/\D/g, ""); if (rutTabla === rut) { let nombre = `${filas[i][2]} ${filas[i][3]} ${filas[i][4]}`.trim(); let curso = filas[i][0]; let puntos = Number(filas[i][5]) || 0; const premios = obtenerPremiosCompletos(puntos); return respuesta(true, "Consulta exitosa", { nombre: nombre, curso: curso, puntos: puntos, premios_disponibles: premios.disponibles, proximo_premio: premios.proximo, puntos_faltantes: premios.faltantes, todos_los_premios: premios.todos }); } } return respuesta(false, "RUT no encontrado"); } function obtenerPremiosCompletos(puntos) { const ss = SpreadsheetApp.getActive(); const hojaPremios = ss.getSheetByName(HOJA_PREMIOS); if (!hojaPremios) return { disponibles: [], proximo: null, faltantes: null, todos: [] }; const premiosData = hojaPremios.getDataRange().getValues(); let disponibles = []; let todos = []; let proximoCosto = null; let proximoNombre = null; for (let i = 1; i < premiosData.length; i++) { let costo = Number(premiosData[i][0]); let nombre = premiosData[i][1]; if (isNaN(costo) || !nombre) continue; todos.push({ costo: costo, nombre: nombre }); if (puntos >= costo) { disponibles.push(nombre); } else { if (proximoCosto === null || costo < proximoCosto) { proximoCosto = costo; proximoNombre = nombre; } } } todos.sort((a,b) => a.costo - b.costo); let faltantes = proximoCosto ? proximoCosto - puntos : null; return { disponibles: disponibles, proximo: proximoNombre, faltantes: faltantes, todos: todos }; } function canjearPremio(data) { const rut = data.rut.toString().replace(/\D/g, ""); const premioNombre = data.premio; const ss = SpreadsheetApp.getActive(); const hojaDatos = ss.getSheetByName(HOJA_DATOS); const hojaPremios = ss.getSheetByName(HOJA_PREMIOS); const hojaCanjes = ss.getSheetByName(HOJA_CANJES); const premiosData = hojaPremios.getDataRange().getValues(); let costo = null; for (let i = 1; i < premiosData.length; i++) { if (premiosData[i][1] === premioNombre) { costo = Number(premiosData[i][0]); break; } } if (!costo) return respuesta(false, "Premio no válido"); const filas = hojaDatos.getDataRange().getValues(); for (let i = 1; i < filas.length; i++) { let rutTabla = filas[i][1].toString().replace(/\D/g, ""); if (rutTabla === rut) { let puntosActual = Number(filas[i][5]) || 0; if (puntosActual >= costo) { let nuevosPuntos = puntosActual - costo; hojaDatos.getRange(i+1, 6).setValue(nuevosPuntos); let nombre = `${filas[i][2]} ${filas[i][3]} ${filas[i][4]}`.trim(); let curso = filas[i][0]; let fecha = new Date(); hojaCanjes.appendRow([fecha, rut, nombre, curso, premioNombre, costo]); return respuesta(true, "Canje exitoso", { nombre: nombre, puntos_restantes: nuevosPuntos, premio: premioNombre, costo: costo }); } else { return respuesta(false, "Puntos insuficientes", { necesitas: costo, tienes: puntosActual }); } } } return respuesta(false, "RUT no encontrado"); } // ================================================== // FUNCIÓN PARA LISTAR CANJES (HISTORIAL) // ================================================== function listarCanjes() { const ss = SpreadsheetApp.getActive(); const hojaCanjes = ss.getSheetByName(HOJA_CANJES); if (!hojaCanjes) return respuesta(false, "Hoja CANJES no encontrada"); const datos = hojaCanjes.getDataRange().getValues(); const canjes = []; for (let i = 1; i < datos.length; i++) { const fila = datos[i]; if (!fila[0]) continue; let fecha = fila[0]; if (fecha instanceof Date) { fecha = Utilities.formatDate(fecha, Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss"); } canjes.push({ fecha: fecha, rut: fila[1]?.toString() || "", nombre: fila[2]?.toString() || "", curso: fila[3]?.toString() || "", premio: fila[4]?.toString() || "", puntos: fila[5]?.toString() || "" }); } canjes.sort((a, b) => b.fecha.localeCompare(a.fecha)); return respuesta(true, "Lista de canjes", { canjes: canjes }); } // ================================================== // FUNCIONES PARA GESTIÓN DE ALUMNOS (CRUD) // ================================================== function listarAlumnos() { const ss = SpreadsheetApp.getActive(); const hoja = ss.getSheetByName(HOJA_DATOS); if (!hoja) return respuesta(false, "Hoja DATOS no encontrada"); const datos = hoja.getDataRange().getValues(); const alumnos = []; for (let i = 1; i < datos.length; i++) { const fila = datos[i]; alumnos.push({ curso: fila[0]?.toString() || "", rut: fila[1]?.toString() || "", nombres: fila[2]?.toString() || "", apellidoPaterno: fila[3]?.toString() || "", apellidoMaterno: fila[4]?.toString() || "", puntos: fila[5]?.toString() || "0" }); } return respuesta(true, "Lista de alumnos", { alumnos: alumnos }); } function agregarAlumno(data) { const { curso, rut, nombres, apellidoPaterno, apellidoMaterno } = data; if (!rut || !nombres || !apellidoPaterno) { return respuesta(false, "Faltan datos obligatorios (RUT, Nombres, Apellido Paterno)"); } const rutLimpio = rut.toString().replace(/\D/g, ""); if (rutLimpio.length < 6) return respuesta(false, "RUT inválido (muy corto)"); const ss = SpreadsheetApp.getActive(); const hoja = ss.getSheetByName(HOJA_DATOS); const filas = hoja.getDataRange().getValues(); for (let i = 1; i < filas.length; i++) { let rutTabla = filas[i][1].toString().replace(/\D/g, ""); if (rutTabla === rutLimpio) { return respuesta(false, "El RUT ya existe"); } } hoja.appendRow([curso, rutLimpio, nombres, apellidoPaterno, apellidoMaterno || "", 0]); return respuesta(true, "Alumno agregado exitosamente"); } function editarAlumno(data) { const { rutOriginal, curso, rut, nombres, apellidoPaterno, apellidoMaterno } = data; if (!rutOriginal) return respuesta(false, "RUT original no proporcionado"); const rutOriginalLimpio = rutOriginal.toString().replace(/\D/g, ""); const rutNuevoLimpio = rut.toString().replace(/\D/g, ""); const ss = SpreadsheetApp.getActive(); const hoja = ss.getSheetByName(HOJA_DATOS); const filas = hoja.getDataRange().getValues(); let filaIndex = -1; for (let i = 1; i < filas.length; i++) { let rutTabla = filas[i][1].toString().replace(/\D/g, ""); if (rutTabla === rutOriginalLimpio) { filaIndex = i + 1; break; } } if (filaIndex === -1) return respuesta(false, "Alumno no encontrado"); if (rutOriginalLimpio !== rutNuevoLimpio) { for (let i = 1; i < filas.length; i++) { let rutTabla = filas[i][1].toString().replace(/\D/g, ""); if (rutTabla === rutNuevoLimpio && (i+1) !== filaIndex) { return respuesta(false, "El nuevo RUT ya existe en otro alumno"); } } } hoja.getRange(filaIndex, 1).setValue(curso); hoja.getRange(filaIndex, 2).setValue(rutNuevoLimpio); hoja.getRange(filaIndex, 3).setValue(nombres); hoja.getRange(filaIndex, 4).setValue(apellidoPaterno); hoja.getRange(filaIndex, 5).setValue(apellidoMaterno || ""); return respuesta(true, "Alumno actualizado exitosamente"); } function eliminarAlumno(data) { const { rut } = data; if (!rut) return respuesta(false, "RUT no proporcionado"); const rutLimpio = rut.toString().replace(/\D/g, ""); const ss = SpreadsheetApp.getActive(); const hoja = ss.getSheetByName(HOJA_DATOS); const filas = hoja.getDataRange().getValues(); let filaIndex = -1; for (let i = 1; i < filas.length; i++) { let rutTabla = filas[i][1].toString().replace(/\D/g, ""); if (rutTabla === rutLimpio) { filaIndex = i + 1; break; } } if (filaIndex === -1) return respuesta(false, "Alumno no encontrado"); hoja.deleteRow(filaIndex); return respuesta(true, "Alumno eliminado exitosamente"); } // ================================================== // NUEVAS FUNCIONES PARA GESTIÓN DE PREMIOS (CRUD) // ================================================== function listarPremios() { const ss = SpreadsheetApp.getActive(); const hoja = ss.getSheetByName(HOJA_PREMIOS); if (!hoja) return respuesta(false, "Hoja PREMIOS no encontrada"); const datos = hoja.getDataRange().getValues(); const premios = []; for (let i = 1; i < datos.length; i++) { const fila = datos[i]; if (!fila[0] && !fila[1]) continue; premios.push({ puntos: fila[0]?.toString() || "", nombre: fila[1]?.toString() || "" }); } return respuesta(true, "Lista de premios", { premios: premios }); } function agregarPremio(data) { const { puntos, nombre } = data; if (!puntos || !nombre) return respuesta(false, "Faltan datos (puntos y nombre)"); const puntosNum = Number(puntos); if (isNaN(puntosNum) || puntosNum <= 0) return respuesta(false, "Puntos inválidos"); const ss = SpreadsheetApp.getActive(); const hoja = ss.getSheetByName(HOJA_PREMIOS); const datos = hoja.getDataRange().getValues(); // Verificar duplicado por nombre (insensible a mayúsculas/minúsculas) for (let i = 1; i < datos.length; i++) { if (datos[i][1]?.toString().toLowerCase() === nombre.toString().toLowerCase()) { return respuesta(false, "Ya existe un premio con ese nombre"); } } hoja.appendRow([puntosNum, nombre]); return respuesta(true, "Premio agregado exitosamente"); } function editarPremio(data) { const { nombreOriginal, puntos, nombre } = data; if (!nombreOriginal || !puntos || !nombre) return respuesta(false, "Faltan datos"); const puntosNum = Number(puntos); if (isNaN(puntosNum) || puntosNum <= 0) return respuesta(false, "Puntos inválidos"); const ss = SpreadsheetApp.getActive(); const hoja = ss.getSheetByName(HOJA_PREMIOS); const datos = hoja.getDataRange().getValues(); let filaIndex = -1; for (let i = 1; i < datos.length; i++) { if (datos[i][1]?.toString() === nombreOriginal) { filaIndex = i + 1; break; } } if (filaIndex === -1) return respuesta(false, "Premio no encontrado"); // Si cambia el nombre, verificar que no exista otro con el nuevo nombre if (nombreOriginal !== nombre) { for (let i = 1; i < datos.length; i++) { if (datos[i][1]?.toString() === nombre && (i+1) !== filaIndex) { return respuesta(false, "Ya existe otro premio con ese nombre"); } } } hoja.getRange(filaIndex, 1).setValue(puntosNum); hoja.getRange(filaIndex, 2).setValue(nombre); return respuesta(true, "Premio actualizado exitosamente"); } function eliminarPremio(data) { const { nombre } = data; if (!nombre) return respuesta(false, "Nombre del premio no proporcionado"); const ss = SpreadsheetApp.getActive(); const hoja = ss.getSheetByName(HOJA_PREMIOS); const datos = hoja.getDataRange().getValues(); let filaIndex = -1; for (let i = 1; i < datos.length; i++) { if (datos[i][1]?.toString() === nombre) { filaIndex = i + 1; break; } } if (filaIndex === -1) return respuesta(false, "Premio no encontrado"); hoja.deleteRow(filaIndex); return respuesta(true, "Premio eliminado exitosamente"); } function respuesta(ok, mensaje, data = {}) { return ContentService.createTextOutput(JSON.stringify({ status: ok ? "ok" : "error", message: mensaje, ...data })).setMimeType(ContentService.MimeType.JSON); }
